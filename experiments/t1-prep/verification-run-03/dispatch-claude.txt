## 1. 任务文本（一次下发，全文逐字）

> 请在当前工作空间内创建纯本地单文件网页应用 `mindmap.html`（HTML/CSS/JS 全部在此文件内）与 `README.md`（打开方式与用法）。文件分三类：**交付物**＝上述两文件；**运行时数据**＝验收时经"导出"产生的 JSON 文件；**开发验证文件**＝你开发中自建的测试或草稿，如需要请放入工作空间内 `dev/` 子目录，它们不作为验收依据。**边界说明**：对联网、外部资源与文件写入的限制针对**交付物（`mindmap.html`）运行时行为**——运行时不得联网、不加载任何外部资源（库/字体/图标/CDN）、无构建步骤与依赖安装，且不得在工作空间之外创建或修改任何文件。功能验收评价交付物行为；开发过程仍须遵守实验约束：不安装依赖、不获取外部库或资源，任务代码、测试和草稿仅写入指定实验工作空间。被测产品客户端的正常模型服务通信及自身必要日志、缓存，不计作交付物联网或任务越界写入，但这一例外不授权读取凭据、修改客户端设置或访问任务无关文件；相关过程行为未被监测时记为未知，不默认判通过。
>
> 功能需求（全部必做）：
> 1. **生成**：页面左侧提供多行文本框和"生成导图"按钮。文本规则：每行一个节点标题；行首空格表示层级，2 个空格为一层；**第一条非空有效行**必须顶格，为根节点；每个后续有效行的缩进必须恰好比其父行多 2 个空格；空白行（零长度或仅空白字符）一律忽略（含 Tab 的行除外，Tab 检查优先）；Tab 为非法字符。不允许多根节点（第二个顶格有效行即非法）。
> 2. **非法输入处理**：点击"生成导图"时若存在非法行（缩进跳层、有效首行不顶格、含 Tab 等），或全部行均为空白行而被忽略后无任何有效行（空输入），必须**整体拒绝**本次生成：显示可见的错误提示，当前导图与撤销记录保持完全不变。
> 3. **生成成功**＝用文本框内容**整体替换**当前导图，并**清空撤销记录**；不需要确认弹窗。仅一行顶格文本为合法输入，生成单节点导图。
> 4. **编辑**（直接作用于导图）：a) 重命名任意节点的标题；b) 为任意节点添加子节点（排在现有子节点末尾）；c) 删除任意节点（连同其全部后代，允许删除根节点＝清空整图）。编辑只影响目标节点及其子树，其余节点的标题、层级、相对顺序必须保持不变。标题统一规范适用于所有来源：去除首尾空白后非空且不含 Tab。产生空标题、含 Tab 的编辑，以及重命名为与原标题完全相同值的编辑，均视为无效编辑：状态不变、不进入撤销记录、不中断不报错。
> 5. **撤销**：提供"撤销"按钮，每点击一次回退最近一次编辑操作（一次重命名/一次添加/一次删除各算一次；删除被撤销时须整体恢复被删子树）；至少支持连续撤销 10 次；撤销到最早记录后再点击须无任何效果且不报错。不要求"重做"。撤销只针对编辑；生成与导入均不可撤销（其效果已由"整体替换＋清栈"定义）。
> 6. **导出与导入**：提供"导出 JSON"按钮：将当前导图保存为符合公开结构 `{"title": 字符串, "children": [同构递归]}`（UTF-8）的 JSON 文件；导出采用浏览器默认下载行为，是否弹出保存对话框取决于浏览器与使用者设置，不作要求（建议文件名 `mindmap.json`）。提供"导入 JSON"按钮：经文件选择对话框显式选择一个 JSON 文件，解析与结构校验通过后**整体替换**当前导图并**清空撤销记录**，无需确认弹窗；使用者取消选择时**静默忽略**（无变化、无提示亦可、不报错）；文件解析失败或结构不符合公开结构时，必须**整体拒绝**：显示可见提示，当前导图与撤销记录完全不变。导入内容按统一标题规范规范化（去 BOM、仅保留 `title`/`children` 结构、标题去首尾空白；任一节点不满足即整体拒绝）。应用不得自动发现或自动加载任何本地文件。
> 7. **明确不做**：重做、自动布局优化、拖拽移动、多根节点、联网同步、账号系统、语音、自动加载本地文件。
>
> 验收只依据上述行为是否正确；界面布局、视觉风格与内部实现方式不限。

| 重复标题与节点识别 | 标题不唯一、不去重不合并；节点身份＝树中位置（验收用"根的第 N 个子节点"式路径定位）；内部标识自由 | 判据必要，实现自由 |

'use strict';
var root = null;        // 节点：{title: string, children: array}
var undoStack = [];     // 撤销记录：{type:'rename'|'add'|'delete', ...}

function $(id){ return document.getElementById(id); }
function showMsg(text, isError){
  var m = $('msg');
  m.textContent = text;
  m.style.color = isError ? '#c22' : '#171';
}

/* ---------- 标题统一规范：原始串不含 Tab（首/中/尾均拒），再去首尾空白且非空 ---------- */
function normalizeTitle(t){
  var s = String(t);
  if (s.indexOf('\t') !== -1) return null;   // 先查原始字符串：首部/中间/尾部 Tab 均拒绝
  s = s.trim();
  if (s.length === 0) return null;
  return s;
}

/* ---------- 输入解析 ---------- */
function parseInput(text){
  var lines = text.split(/\r\n|\r|\n/);
  var items = [];   // {indent, title, line}
  for (var i = 0; i < lines.length; i++){
    var raw = lines[i];
    if (raw.length === 0) continue;                       // 零长度空行忽略
    if (raw.indexOf('\t') !== -1) return {error:'第 '+(i+1)+' 行：含 Tab（非法字符）'};
    if (/^\s+$/.test(raw)) continue;                      // 仅空白字符的行忽略（Tab 已在上一步拦截）
    var mLead = raw.match(/^ +/);
    var indent = mLead ? mLead[0].length : 0;
    var title = normalizeTitle(raw.slice(indent));
    if (title === null) return {error:'第 '+(i+1)+' 行：空标题或含 Tab'};
    if (indent % 2 !== 0) return {error:'第 '+(i+1)+' 行：缩进不是 2 空格的整数倍'};
    items.push({indent: indent, title: title, line: i+1});
  }
  if (items.length === 0) return {error:'输入为空：至少需要一行根节点'};
  if (items[0].indent !== 0) return {error:'第 '+items[0].line+' 行：首行必须顶格'};
  var rootNode = {title: items[0].title, children: []};
  var stack = [{indent: 0, node: rootNode}];
  for (var k = 1; k < items.length; k++){
    var it = items[k];
    while (stack.length > 1 && stack[stack.length-1].indent >= it.indent) stack.pop();
    var parent = stack[stack.length-1];
    if (it.indent !== parent.indent + 2) return {error:'第 '+it.line+' 行：缩进不符合层级规则（多根或跳层）'};
    var node = {title: it.title, children: []};
    parent.node.children.push(node);
    stack.push({indent: it.indent, node: node});
  }
  return {root: rootNode};
}

/* ---------- 渲染 ---------- */
function render(){
  var map = $('map');
  map.innerHTML = '';
  if (!root){
    var d = document.createElement('div');
    d.className = 'empty';
    d.textContent = '（空）尚未生成导图。请在左侧输入缩进文本后点击“生成导图”，或“导入 JSON”。';
    map.appendChild(d);
    return;
  }
  var ul = document.createElement('ul');
  ul.appendChild(renderNode(root, null, -1));
  map.appendChild(ul);
}
function renderNode(node, parentNode, indexInParent){
  var li = document.createElement('li');
  var row = document.createElement('div');
  row.className = 'node-row' + (parentNode === null ? ' rootrow' : '');
  var title = document.createElement('span');
  title.className = 'title';
  title.textContent = node.title;
  row.appendChild(title);
  row.appendChild(mkBtn('重命名', function(){ doRename(node); }));
  row.appendChild(mkBtn('+子节点', function(){ doAddChild(node); }));
  row.appendChild(mkBtn('删除', function(){ doDelete(node, parentNode, indexInParent); }));
  li.appendChild(row);
  if (node.children.length > 0){
    var ul = document.createElement('ul');
    for (var i = 0; i < node.children.length; i++){
      ul.appendChild(renderNode(node.children[i], node, i));
    }
    li.appendChild(ul);
  }
  return li;
}
function mkBtn(label, fn){
  var b = document.createElement('button');
  b.textContent = label;
  b.onclick = fn;
  return b;
}

/* ---------- 编辑（无效编辑：状态不变、不进栈、不中断） ---------- */
function doRename(node){
  var t = window.prompt('新的标题（取消、空标题、含 Tab 或与原标题相同＝放弃本次编辑）：', node.title);
  if (t === null) return;
  var nt = normalizeTitle(t);
  if (nt === null){ showMsg('重命名被拒绝：标题为空或含 Tab', true); return; }
  if (nt === node.title){ showMsg('重命名被拒绝：标题未变化（无效编辑，不进入撤销记录）', true); return; }
  undoStack.push({type:'rename', node: node, oldTitle: node.title});
  node.title = nt;
  showMsg('已重命名', false);
  render();
}
function doAddChild(parentNode){
  var t = window.prompt('子节点标题（取消或空标题＝放弃本次编辑）：', '');
  if (t === null) return;
  var nt = normalizeTitle(t);
  if (nt === null){ showMsg('添加被拒绝：标题为空或含 Tab', true); return; }
  var child = {title: nt, children: []};
  undoStack.push({type:'add', parentNode: parentNode, child: child});
  parentNode.children.push(child);
  showMsg('已添加子节点', false);
  render();
}
function doDelete(node, parentNode, indexInParent){
  undoStack.push({type:'delete', parentNode: parentNode, node: node, index: indexInParent});
  if (parentNode === null){ root = null; }
  else { parentNode.children.splice(indexInParent, 1); }
  showMsg('已删除', false);
  render();
}

/* ---------- 撤销 ---------- */
function doUndo(){
  if (undoStack.length === 0){ showMsg('没有可撤销的操作', false); return; }
  var op = undoStack.pop();
  if (op.type === 'rename'){ op.node.title = op.oldTitle; }
  else if (op.type === 'add'){
    var i = op.parentNode.children.indexOf(op.child);
    if (i !== -1) op.parentNode.children.splice(i, 1);
  }
  else if (op.type === 'delete'){
    if (op.parentNode === null){ root = op.node; }
    else { op.parentNode.children.splice(op.index, 0, op.node); }
  }
  showMsg('已撤销一步（剩余可撤销 ' + undoStack.length + ' 步）', false);
  render();
}

$('file-in').addEventListener('change', function(ev){
  var f = ev.target.files && ev.target.files[0];
  if (!f){ return; }   // 取消：静默忽略，无变化、不提示、不报错
  var reader = new FileReader();
  reader.onload = function(){
    var txt = String(reader.result);
    if (txt.charCodeAt(0) === 0xFEFF) txt = txt.slice(1);   // 规范化：去 BOM
    var obj = null;
    try { obj = JSON.parse(txt); }
    catch (e){ showMsg('导入被拒绝：文件不是合法 JSON；当前导图与撤销记录不变', true); return; }
    var tree = normalizeTree(obj);
    if (tree === null){ showMsg('导入被拒绝：结构或标题不符合公开结构 {title, children}；当前导图与撤销记录不变', true); return; }
    root = tree;
    undoStack = [];
    showMsg('导入成功：导图已整体替换，撤销记录已清空', false);
    render();
  };
  reader.onerror = function(){ showMsg('导入被拒绝：文件读取失败；当前导图与撤销记录不变', true); };
  reader.readAsText(f, 'utf-8');
});
function normalizeTree(n){
  if (typeof n !== 'object' || n === null || Array.isArray(n)) return null;
  if (typeof n.title !== 'string') return null;
  if (!Array.isArray(n.children)) return null;
  var t = normalizeTitle(n.title);
  if (t === null) return null;
  var node = {title: t, children: []};   // 规范化：仅保留 title/children，多余字段忽略
  for (var i = 0; i < n.children.length; i++){
    var c = normalizeTree(n.children[i]);
    if (c === null) return null;
    node.children.push(c);
  }
  return node;
}
