def dialog_process_ok(dp, browser_pid, procs, browser_exe_name):
    # dialog hosted by browser itself, or by a chrome.exe descendant of browser_pid
    if dp == browser_pid: return True
    if not browser_exe_name: return False
    cur = dp
    seen = 0
    while cur in procs and seen < 64:
        ppid, exe = procs[cur]
        if ppid == browser_pid: return True
        if (exe or "").lower() != browser_exe_name: return False
        cur = ppid
        seen += 1
    return False
